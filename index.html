def score_context(source, answer):
    support = semantic_support(source, answer)
    risk = injection_likelihood(source)
    delta = answer_prob() - intervene(source).answer_prob()

    return {
        "safe": support > 0.72 and risk < 0.18,
        "causal_delta": round(delta, 4),
        "action": "route" if delta > 0.21 else "keep"
    }
